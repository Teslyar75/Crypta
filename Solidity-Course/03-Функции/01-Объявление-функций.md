# Функции в Solidity

## 📋 Объявление функции

```solidity
function functionName(параметры) visibility modifiers returns (тип) {
    // тело функции
}
```

## 🔍 Модификаторы видимости

### public
Доступна всем (внутри и снаружи контракта)

```solidity
function publicFunction() public returns (uint) {
    return 42;
}
```

### private
Только внутри текущего контракта

```solidity
function _privateFunction() private returns (uint) {
    return 42;
}
```

### internal
Внутри контракта и наследников

```solidity
function internalFunction() internal returns (uint) {
    return 42;
}
```

### external

Только снаружи контракта

```solidity
function externalFunction() external returns (uint) {
    return 42;
}
```

## 🔒 Модификаторы состояния

### view
Читает состояние, но не изменяет

```solidity
uint public data = 100;

function getData() public view returns (uint) {
    return data;  // Только чтение
}
```

### pure
Не читает и не изменяет состояние

```solidity
function add(uint a, uint b) public pure returns (uint) {
    return a + b;  // Только вычисления
}
```

### payable
Может принимать эфир

```solidity
function deposit() public payable {
    // msg.value - количество отправленного эфира
}
```

## 📤 Возвращаемые значения

### Один возврат

```solidity
function getNumber() public pure returns (uint) {
    return 42;
}
```

### Множественный возврат

```solidity
function getValues() public pure returns (uint, bool, string memory) {
    return (42, true, "Hello");
}

// Использование
function useValues() public pure {
    (uint num, bool flag, string memory text) = getValues();
}
```

### Именованные возвраты

```solidity
function calculate() public pure returns (uint sum, uint product) {
    sum = 5 + 3;
    product = 5 * 3;
    // return не обязателен
}
```

## 🎯 Специальные переменные

```solidity
contract SpecialVariables {
    function getInfo() public payable returns (address, uint, uint) {
        address sender = msg.sender;      // Адрес отправителя
        uint value = msg.value;           // Количество wei
        uint gasLeft = gasleft();         // Оставшийся газ
        bytes memory data = msg.data;     // Данные вызова
        
        uint blockNum = block.number;     // Номер блока
        uint timestamp = block.timestamp; // Время блока
        
        return (sender, value, gasLeft);
    }
}
```

## 🔄 Модификаторы функций

```solidity
contract ModifierExample {
    address public owner;
    
    constructor() {
        owner = msg.sender;
    }
    
    // Объявление модификатора
    modifier onlyOwner() {
        require(msg.sender == owner, "Not owner");
        _;  // Выполнение функции
    }
    
    // Использование модификатора
    function restrictedFunction() public onlyOwner {
        // Только owner может вызвать
    }
}
```
