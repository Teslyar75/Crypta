# Наследование в Solidity

## 🧬 Базовое наследование

```solidity
// Базовый контракт
contract Animal {
    string public name;
    
    constructor(string memory _name) {
        name = _name;
    }
    
    function speak() public virtual returns (string memory) {
        return "Some sound";
    }
}

// Наследование
contract Dog is Animal {
    constructor(string memory _name) Animal(_name) {}
    
    // Переопределение функции
    function speak() public pure override returns (string memory) {
        return "Woof!";
    }
}
```

## 🔄 Множественное наследование

```solidity
contract A {
    function foo() public virtual returns (string memory) {
        return "A";
    }
}

contract B {
    function bar() public virtual returns (string memory) {
        return "B";
    }
}

// Множественное наследование
contract C is A, B {
    // Наследует foo() и bar()
}
```

## 🎯 Абстрактные контракты

```solidity
abstract contract Shape {
    // Абстрактная функция (без реализации)
    function area() public virtual returns (uint);
    
    // Обычная функция
    function description() public pure returns (string memory) {
        return "This is a shape";
    }
}

contract Circle is Shape {
    uint public radius;
    
    constructor(uint _radius) {
        radius = _radius;
    }
    
    // Обязательная реализация
    function area() public view override returns (uint) {
        return 3 * radius * radius;
    }
}
```

## 📋 Интерфейсы

```solidity
interface IERC20 {
    function totalSupply() external view returns (uint256);
    function balanceOf(address account) external view returns (uint256);
    function transfer(address to, uint256 amount) external returns (bool);
}

contract MyToken is IERC20 {
    // Реализация всех функций интерфейса
    mapping(address => uint256) private balances;
    uint256 private _totalSupply;
    
    function totalSupply() external view override returns (uint256) {
        return _totalSupply;
    }
    
    function balanceOf(address account) external view override returns (uint256) {
        return balances[account];
    }
    
    function transfer(address to, uint256 amount) external override returns (bool) {
        balances[msg.sender] -= amount;
        balances[to] += amount;
        return true;
    }
}
```
