# Типы данных в Solidity

## 📊 Обзор типов

Solidity - **статически типизированный** язык. Тип каждой переменной должен быть указан явно.

## 🔢 Числовые типы

### Целые числа (Integer)

#### uint (беззнаковые)

```solidity
uint8   // 0 до 255 (2^8 - 1)
uint16  // 0 до 65535 (2^16 - 1)
uint32  // 0 до 4294967295 (2^32 - 1)
uint256 // 0 до 2^256 - 1 (по умолчанию)

// uint = uint256
uint public myNumber = 42;
```

#### int (знаковые)

```solidity
int8    // -128 до 127
int16   // -32768 до 32767
int256  // -2^255 до 2^255 - 1

int public temperature = -15;
```

### Операции с числами

```solidity
contract MathOperations {
    function calculate() public pure returns (uint) {
        uint a = 10;
        uint b = 3;
        
        uint sum = a + b;        // 13
        uint diff = a - b;       // 7
        uint product = a * b;    // 30
        uint quotient = a / b;   // 3 (целочисленное деление!)
        uint remainder = a % b;  // 1 (остаток)
        
        return sum;
    }
}
```

**⚠️ Важно**: В Solidity 0.8+ есть автоматическая проверка переполнения!

```solidity
// До 0.8.0 - переполнение без ошибки
// С 0.8.0+ - ошибка при переполнении
uint8 x = 255;
x = x + 1; // Ошибка! Overflow
```

## 📍 Адреса (Address)

**address** - специальный тип для адресов Ethereum (20 байт).

```solidity
contract AddressExample {
    address public owner;
    address payable public recipient;
    
    constructor() {
        owner = msg.sender; // Адрес отправителя
    }
    
    // Проверка баланса
    function getBalance(address addr) public view returns (uint) {
        return addr.balance;
    }
    
    // Отправка эфира (только для payable)
    function sendEther(address payable to) public payable {
        to.transfer(msg.value);
    }
}
```

### Методы address

```solidity
address addr = 0x1234567890123456789012345678901234567890;

addr.balance        // Баланс в wei
addr.code          // Байт-код контракта
addr.codehash      // Хеш кода

// Для payable address:
payable(addr).transfer(amount)  // Отправить ETH
payable(addr).send(amount)      // Отправить ETH (возвращает bool)
```

## ✅ Булевы значения (Boolean)

```solidity
bool public isActive = true;
bool public isCompleted = false;

function checkCondition() public pure returns (bool) {
    bool a = true;
    bool b = false;
    
    bool and = a && b;  // false (И)
    bool or = a || b;   // true (ИЛИ)
    bool not = !a;      // false (НЕ)
    
    return and;
}
```

## 📝 Строки и байты

### Строки (String)

```solidity
string public name = "Alice";
string public greeting = "Hello, World!";

function concatenate() public pure returns (string memory) {
    // Конкатенация строк
    return string(abi.encodePacked("Hello, ", "Blockchain!"));
}
```

**⚠️ Ограничения**:
- Нельзя получить длину напрямую
- Нельзя обращаться по индексу
- Дорого по газу

### Байты (Bytes)

#### Фиксированного размера

```solidity
bytes1 public b1 = 0x12;
bytes2 public b2 = 0x1234;
bytes32 public b32; // Часто используется для хешей

function getHash() public pure returns (bytes32) {
    return keccak256(abi.encodePacked("Hello"));
}
```

#### Динамического размера

```solidity
bytes public data;

function setData() public {
    data = "Hello";
    uint len = data.length;  // Можно получить длину
    bytes1 first = data[0];  // Можно обращаться по индексу
}
```

## 📅 Перечисления (Enum)

```solidity
contract EnumExample {
    // Объявление enum
    enum Status {
        Pending,    // 0
        Approved,   // 1
        Rejected,   // 2
        Completed   // 3
    }
    
    Status public currentStatus;
    
    constructor() {
        currentStatus = Status.Pending;
    }
    
    function approve() public {
        currentStatus = Status.Approved;
    }
    
    function getStatus() public view returns (Status) {
        return currentStatus;
    }
}
```

## 🏗️ Структуры (Struct)

```solidity
contract StructExample {
    struct Person {
        string name;
        uint age;
        address wallet;
    }
    
    Person public alice;
    
    function createPerson() public {
        // Способ 1: По именам полей
        alice = Person({
            name: "Alice",
            age: 25,
            wallet: msg.sender
        });
        
        // Способ 2: По порядку
        alice = Person("Alice", 25, msg.sender);
    }
    
    function getAge() public view returns (uint) {
        return alice.age;
    }
}
```

## 📦 Массивы (Arrays)

### Фиксированного размера

```solidity
uint[5] public fixedArray;  // Массив из 5 элементов

function setValues() public {
    fixedArray = [1, 2, 3, 4, 5];
}
```

### Динамического размера

```solidity
uint[] public dynamicArray;

function addElement(uint value) public {
    dynamicArray.push(value);  // Добавить элемент
}

function getLength() public view returns (uint) {
    return dynamicArray.length;
}

function removeLastElement() public {
    dynamicArray.pop();  // Удалить последний
}
```

## 🗺️ Mapping (Словари)

```solidity
contract MappingExample {
    // Ключ => Значение
    mapping(address => uint) public balances;
    mapping(address => bool) public isRegistered;
    
    function setBalance(uint amount) public {
        balances[msg.sender] = amount;
    }
    
    function getBalance(address user) public view returns (uint) {
        return balances[user];  // По умолчанию 0
    }
    
    // Вложенные mapping
    mapping(address => mapping(address => uint)) public allowances;
}
```

**⚠️ Особенности mapping**:
- Нельзя итерировать
- Нельзя получить все ключи
- Все ключи существуют (значение по умолчанию)
- Только в storage

## 💾 Расположение данных

### Storage (постоянное хранилище)

```solidity
uint public storedData;  // Хранится в блокчейне
```

### Memory (временное хранилище)

```solidity
function processData(string memory text) public pure {
    // text существует только во время выполнения функции
}
```

### Calldata (неизменяемое временное)

```solidity
function processData(string calldata text) external pure {
    // Дешевле memory, но нельзя изменить
}
```
