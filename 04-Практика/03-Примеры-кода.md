## 🐍 Python примеры

### 1. Простой блокчейн

```python
import hashlib
import json
from time import time

class Block:
    def __init__(self, index, transactions, timestamp, previous_hash, nonce=0):
        self.index = index
        self.transactions = transactions
        self.timestamp = timestamp
        self.previous_hash = previous_hash
        self.nonce = nonce
    
    def compute_hash(self):
        """Вычисление хеша блока"""
        block_string = json.dumps(self.__dict__, sort_keys=True)
        return hashlib.sha256(block_string.encode()).hexdigest()

class Blockchain:
    difficulty = 2  # Количество нулей в начале хеша
    
    def __init__(self):
        self.chain = []
        self.pending_transactions = []
        # Создание genesis блока
        self.create_genesis_block()
    
    def create_genesis_block(self):
        """Создание первого блока"""
        genesis_block = Block(0, [], time(), "0")
        genesis_block.hash = genesis_block.compute_hash()
        self.chain.append(genesis_block)
    
    def get_last_block(self):
        return self.chain[-1]
    
    def proof_of_work(self, block):
        """Proof of Work алгоритм"""
        block.nonce = 0
        computed_hash = block.compute_hash()
        
        while not computed_hash.startswith('0' * Blockchain.difficulty):
            block.nonce += 1
            computed_hash = block.compute_hash()
        
        return computed_hash
    
    def add_block(self, block, proof):
        """Добавление блока в цепочку"""
        previous_hash = self.get_last_block().hash
        
        if previous_hash != block.previous_hash:
            return False
        
        if not self.is_valid_proof(block, proof):
            return False
        
        block.hash = proof
        self.chain.append(block)
        return True
    
    def is_valid_proof(self, block, block_hash):
        """Проверка Proof of Work"""
        return (block_hash.startswith('0' * Blockchain.difficulty) and
                block_hash == block.compute_hash())
    
    def add_transaction(self, transaction):
        """Добавление транзакции"""
        self.pending_transactions.append(transaction)
    
    def mine(self):
        """Майнинг нового блока"""
        if not self.pending_transactions:
            return False
        
        last_block = self.get_last_block()
        
        new_block = Block(
            index=last_block.index + 1,
            transactions=self.pending_transactions,
            timestamp=time(),
            previous_hash=last_block.hash
        )
        
        proof = self.proof_of_work(new_block)
        self.add_block(new_block, proof)
        self.pending_transactions = []
        
        return new_block.index

# Использование
blockchain = Blockchain()

# Добавление транзакций
blockchain.add_transaction({"from": "Alice", "to": "Bob", "amount": 50})
blockchain.add_transaction({"from": "Bob", "to": "Charlie", "amount": 25})

# Майнинг блока
print("Майним блок...")
blockchain.mine()

# Просмотр блокчейна
for block in blockchain.chain:
    print(f"Блок #{block.index}")
    print(f"Транзакции: {block.transactions}")
    print(f"Хеш: {block.hash}")
    print(f"Предыдущий хеш: {block.previous_hash}\n")
```

### 2. Генерация Bitcoin адреса

```python
import hashlib
import ecdsa
import base58

def generate_bitcoin_address():
    """Генерация Bitcoin адреса"""
    
    # 1. Генерация приватного ключа (256 бит)
    private_key = hashlib.sha256(b"random seed").digest()
    print(f"Приватный ключ: {private_key.hex()}")
    
    # 2. Генерация публичного ключа (ECDSA)
    sk = ecdsa.SigningKey.from_string(private_key, curve=ecdsa.SECP256k1)
    vk = sk.get_verifying_key()
    public_key = b'\x04' + vk.to_string()
    print(f"Публичный ключ: {public_key.hex()}")
    
    # 3. SHA-256 хеширование
    sha256_hash = hashlib.sha256(public_key).digest()
    
    # 4. RIPEMD-160 хеширование
    ripemd160 = hashlib.new('ripemd160')
    ripemd160.update(sha256_hash)
    hash160 = ripemd160.digest()
    
    # 5. Добавление версии (0x00 для mainnet)
    versioned_hash = b'\x00' + hash160
    
    # 6. Контрольная сумма (первые 4 байта двойного SHA-256)
    checksum = hashlib.sha256(hashlib.sha256(versioned_hash).digest()).digest()[:4]
    
    # 7. Финальный адрес (Base58 кодирование)
    address_bytes = versioned_hash + checksum
    address = base58.b58encode(address_bytes).decode()
    
    print(f"Bitcoin адрес: {address}")
    
    return private_key, public_key, address

# Генерация адреса
generate_bitcoin_address()
```

### 3. Проверка баланса (через API)

```python
import requests

def get_balance(address):
    """Получение баланса Bitcoin адреса"""
    url = f"https://blockchain.info/q/addressbalance/{address}"
    
    try:
        response = requests.get(url)
        balance_satoshi = int(response.text)
        balance_btc = balance_satoshi / 100000000
        
        print(f"Адрес: {address}")
        print(f"Баланс: {balance_btc} BTC ({balance_satoshi} satoshi)")
        
        return balance_btc
    except Exception as e:
        print(f"Ошибка: {e}")
        return None

# Пример (адрес создателя Bitcoin)
get_balance("1A1zP1eP5QGefi2DMPTfTL5SLmv7DivfNa")
```

### 4. Создание и проверка подписи

```python
import hashlib
import ecdsa

def sign_message(private_key, message):
    """Подписание сообщения"""
    sk = ecdsa.SigningKey.from_string(private_key, curve=ecdsa.SECP256k1)
    message_hash = hashlib.sha256(message.encode()).digest()
    signature = sk.sign_digest(message_hash, sigencode=ecdsa.util.sigencode_der)
    return signature

def verify_signature(public_key, message, signature):
    """Проверка подписи"""
    vk = ecdsa.VerifyingKey.from_string(public_key[1:], curve=ecdsa.SECP256k1)
    message_hash = hashlib.sha256(message.encode()).digest()
    
    try:
        vk.verify_digest(signature, message_hash, sigdecode=ecdsa.util.sigdecode_der)
        return True
    except:
        return False

# Пример использования
private_key = hashlib.sha256(b"my secret").digest()
sk = ecdsa.SigningKey.from_string(private_key, curve=ecdsa.SECP256k1)
public_key = b'\x04' + sk.get_verifying_key().to_string()

message = "Hello, Blockchain!"
signature = sign_message(private_key, message)

print(f"Сообщение: {message}")
print(f"Подпись: {signature.hex()}")
print(f"Подпись валидна: {verify_signature(public_key, message, signature)}")
```
